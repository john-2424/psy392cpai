\begin{frame}{When worlds change, how do agents adapt?}
  \begin{columns}[T, onlytextwidth]
    \column{0.32\textwidth}
      \centering
      \textbf{Reward moves}\\[0.6em]
      {\Large $\star \rightarrow \star$}\\[0.6em]
      Goal location shifts.\\ Dynamics unchanged.
    \column{0.32\textwidth}
      \centering
      \textbf{Wall appears}\\[0.6em]
      {\Large $\square\mspace{-14mu}{\rule{0.08em}{1.2em}}$}\\[0.6em]
      Transition graph changes.\\ Reward unchanged.
    \column{0.32\textwidth}
      \centering
      \textbf{Observation shifts}\\[0.6em]
      {\Large \textcolor{gray!60}{\textbullet}\,\textcolor{gray!60}{\textbullet}\,\textcolor{gray!60}{\textbullet}}\\[0.6em]
      Pixels differ, latent state same.\\ (Hippocampal remapping.)
  \end{columns}
  \vspace{1em}
  \begin{alertblock}{Question}
    Do three standard deep-RL architectures show \emph{different} adaptation signatures along these three axes, as neuroscience predicts?
  \end{alertblock}
\end{frame}

% ============================================================
% Slide 3 -- Neuroscience anchor
% ============================================================
\begin{frame}{Neuroscience anchor: two key frameworks}
  \begin{columns}[T]
    \column{0.5\textwidth}
      \textbf{Momennejad et al., 2017}\\
      \emph{Nat.\ Hum.\ Behav.}
      \begin{itemize}
        \item Humans rapidly revalue after \emph{reward} change.
        \item Slower after \emph{transition} change.
        \item Predicted by the Successor Representation: $Q = \langle\psi, \mathbf{w}\rangle$.
        \item Reward revaluation $=$ update $\mathbf{w}$ only.
      \end{itemize}
    \column{0.5\textwidth}
      \textbf{Sanders, Wilson \& Gershman, 2020}\\
      \emph{eLife}
      \begin{itemize}
        \item Hippocampal remapping as \emph{hidden-state inference}.
        \item \textbf{Rate} remapping: graded change under same state.
        \item \textbf{Global} remapping: obs$\rightarrow$state map replaced.
      \end{itemize}
  \end{columns}
  \vspace{0.8em}
  \textbf{Our job:} build a gridworld where these three axes are \emph{cleanly separable}, then see what each agent does.
\end{frame}

% ============================================================
% Slide 4 -- RL recap
% ============================================================
\begin{frame}{RL recap (Tutorial 05): state, action, reward, policy, value}
  \begin{columns}[T, onlytextwidth]
    \column{0.55\textwidth}
      \begin{itemize}
        \item \textbf{State} $s$: where the agent is (here: an 8$\times$8 image).
        \item \textbf{Action} $a$: up/down/left/right.
        \item \textbf{Reward} $r$: $+1$ at goal, $-0.01$ per step.
        \item \textbf{Policy} $\pi(a|s)$: what to do.
        \item \textbf{Value} $Q(s,a)$: expected future reward.
      \end{itemize}
      \[
        Q(s,a) = \mathbb{E}\Big[\sum_{t=0}^{\infty}\gamma^t r_t \,\Big|\, s_0{=}s, a_0{=}a\Big]
      \]
    \column{0.45\textwidth}
      \textbf{CNN (Tutorial 08) feeds everything.}\\[0.5em]
      Pixel image $\rightarrow$ conv layers $\rightarrow$ feature vector $\phi(s)$.\\[0.6em]
      All three agents share this encoder. They differ only in what they do with $\phi(s)$.
  \end{columns}
\end{frame}

% ============================================================
% Slide 6 -- SR intuition
% ============================================================
\begin{frame}{Successor Representation: factorized value}
  \begin{columns}[T, onlytextwidth]
    \column{0.55\textwidth}
      \textbf{Standard TD:} learn $Q(s,a)$ directly.\\
      \textbf{SR:} factorize it.
      \[
        \boxed{\; Q(s,a) = \langle \psi(s,a),\; \mathbf{w}\rangle \;}
      \]
      \begin{itemize}\setlength\itemsep{0.2em}
        \item $\psi(s,a)$: expected discounted future features. \emph{Reward-agnostic.}
        \item $\mathbf{w}$: reward weights.
        \item If reward changes $\rightarrow$ update $\mathbf{w}$ only, leaving $\psi$ fixed.
      \end{itemize}
    \column{0.45\textwidth}
      \textbf{Why care?}\\[0.3em]
      Momennejad 2017 predicts SR-style agents should \emph{specifically} accelerate at reward revaluation, compared to model-free baselines.\\[0.6em]
      We will test the ``$\mathbf{w}$-only'' variant as a direct analog of the human experiment.
  \end{columns}
\end{frame}

% Slide 9 -- Protocol & metrics
% ============================================================
\begin{frame}{Protocol and metrics}
  \begin{columns}[T, onlytextwidth]
    \column{0.55\textwidth}
      \textbf{Two phases per (agent, seed):}
      \begin{enumerate}\setlength\itemsep{0.1em}
        \item Train on \texttt{stable} to convergence.
        \item \textbf{Zero-shot eval} on all 5 conditions.
        \item \textbf{Few-shot adaptation} on each changed condition (20 PPO batches / 60 SR,Replay episodes).
      \end{enumerate}
      3 seeds $\times$ 3 agents $\times$ 5 conditions.
    \column{0.45\textwidth}
      \textbf{Four metrics:}
      \begin{itemize}\setlength\itemsep{0.1em}
        \item Greedy-success rate (snapshot).
        \item \textbf{AUC} of success over adaptation window.
        \item \textbf{$t_{0.5}$}: first step success $\geq 0.5$.
        \item \textbf{Linear probe} on CNN features (agent pos, goal id, distance).
      \end{itemize}
  \end{columns}
\end{frame}

% Slide 13 -- Diagnosis
% ============================================================
\begin{frame}{Diagnosis -- deep-SF policy-extraction fragility}
  \textbf{What we ruled out:}
  \begin{itemize}\setlength\itemsep{0.1em}
    \item Training budget ($300\rightarrow500$ episodes, still regresses).
    \item Representation collapse (Lehnert 2024 ablation replicates; patched run still shows regression).
    \item Loss weighting (rebalanced $\alpha_r$ $20\rightarrow5$: better probe, same policy problem).
  \end{itemize}
  \vspace{0.5em}
  \textbf{What the probe tells us:} SR's encoder compresses \emph{agent column}. The optimal $(6,1)\rightarrow(1,6)$ path needs column discrimination $\rightarrow$ Q-values tie $\rightarrow$ transient 1.0 peaks, not persistent policies.
  \vspace{0.5em}
  \begin{alertblock}{Concrete fixes (out of scope for this project)}
    \begin{itemize}\setlength\itemsep{0em}
      \item Action-conditioned features $\phi(s,a)$ (Barreto 2017, Lehnert 2024 \S 3).
      \item Explicit Q-margin regularizer.
    \end{itemize}
  \end{alertblock}
\end{frame}

% Slide 14 -- Takeaways
% ============================================================
\begin{frame}{Takeaways}
  \begin{itemize}\setlength\itemsep{0.3em}
    \item \textbf{Five-condition testbed} cleanly separates reward / transition / observation change.
    \item \textbf{H2, H4, H5 supported cleanly.} H1 supported via AUC; H3 not.
    \item \textbf{Representation probe as diagnostic:} turned an ambiguous SR result into an architectural finding.
    \item \textbf{Honest negative:} SR revaluation mechanism works; deep-SF policy extraction is the binding constraint.
    \item \textbf{Future work:} action-conditioned $\phi(s,a)$; Q-margin term; adaptation-phase LR tuning for PPO.
  \end{itemize}
